   279	  if (head === "default") {
   280	    const typed = grammar === PatternGrammar.TYPED_PARAMETER;
   281	    if (grammar === PatternGrammar.BINDING_PATTERN) walkPatternDefaults(pattern.values[1], out, inScope, grammar);
   282	    const value = pattern.values[typed ? 3 : 2];
   283	    if (value) walk(value, out, inScope);
   284	    return;
   285	  }

   288	    walkPatternDefaults(pattern.values[typed ? 2 : 1], out, inScope, grammar);
   289	    return;
   290	  }
   291	  if (head !== "array" && head !== "object") return;
